class User:
    _static_id: int = 0
        
    def __init__(self, ud_file: Optional[str], habits: Optional[list[str]]):
        type(self)._static_id += 1
        self.id: int = type(self)._static_id
        
        self.user_file: Path = ud_file
        self.user_data: pd.DataFrame = None
        self.user_habits: list[Habit] = []
        
        if habits is not None:
            for habit in habits:
                current_habit = Habit(habit, assoc_mqtt_topic=f"{self.id}/provider/{habit}")
                if ud_file is not None:
                    current_habit.extract_weekly_averages(pd.read_csv(ud_file))
                self.user_habits.append(current_habit)
        
        with get_connection() as conn:
            reset_habit_tables(conn)
            for habit in habits:
                conn.execute(
                    """
                    INSERT INTO habits (user_id, job_id, name, is_device)
                    VALUES (?, ?, ?, ?);
                    """,
                    (
                        self.id,
                        "Not Implemented",
                        habit,
                        0
                ),
                )
                
        if ud_file is None:
                self.user_file = Path(f"data/{self.id}_habit_log.csv")
                if not self.user_file.exists():
                    with open(self.user_file, 'w') as csv:
                        csv.write(DATA_FORMAT)
                
    def print_user_mqtt_topics(self):
        return [h.assoc_mqtt_topic for h in self.user_habits]
                     
    def print_habit_rules(self, habit_names: list[str] | str = None):
        if habit_names is None:
            return [
                {
                    "habit_name": h.habit_name,
                    "is_device": h.is_device,
                    "rules": [asdict(r) for r in h.daily_rules]
                }
                for h in self.user_habits
            ]
        else:
            temp_habit_list: list[Habit] = []
            for i, habit in enumerate(self.user_habits):
                if habit.habit_name in habit_names or habit.habit_name is habit_names:
                    temp_habit_list.append(habit)
            return [
                {
                    "habit_name": h.habit_name,
                    "is_device": h.is_device,
                    "rules": [asdict(r) for r in h.daily_rules]
                }
                for h in temp_habit_list
            ]
            
    def _get_user_habits(self):
        return [h.habit_name for h in self.user_habits]
    
    def print_user_habits(self):
        return [habit.habit_name for habit in self.user_habits]
            
    def add_user_habit(self, habit_name: str):
        self.user_habits.append(Habit(habit_name))
    
    def _log_user_habit(self, habit: str):
        timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        with open(self.user_file, "a", newline="", encoding="utf-8") as f:
            writer = csv.writer(f)
            writer.writerow([timestamp, habit, "on", True, True])
        
        with get_connection() as conn:
            # print(datetime.now().isoformat()) is this better to use?
            conn.execute(
                """
                INSERT INTO habit_logs (user_id, habit_name, timestamp, state, self_reported)
                VALUES (?, ?, ?, ?, ?)
                """,
                (
                    self.id,
                    habit,
                    timestamp,
                    "on",
                    True,
            ),
        )
    
    def update_user_habits(self):
        for habit in self.user_habits:
            habit.extract_weekly_averages(pd.read_csv(self.user_file))

    def extract_weekly_averages(self, df):
        self.daily_rules.clear()
        
        df = df[df['habit'] == self.habit_name]
        df['timestamp'] = pd.to_datetime(df['timestamp'])
        df['day_of_week'] = df['timestamp'].dt.dayofweek
        for i in range(0, 7):
            day = df[df['day_of_week'] == i]
        
            if day.empty:
                continue
        
            avg_hour = (int)(day['timestamp'].dt.hour.mean())
            avg_minutes = (int)(day['timestamp'].dt.minute.mean())
        
            self.daily_rules.append(Rule(i, avg_hour, avg_minutes, True))
